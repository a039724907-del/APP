/*
 ============================================================================
  מדריך קצר - 5 מושגים שמופיעים בקוד (קראו פעם אחת ואז הכל יהיה ברור)
 ============================================================================

 1) משתנה
    שם שמתחיל ב-$ ומחזיק ערך.  $phone = "0501234567";

 2) מערך (array)
    רשימה של דברים בתוך סוגריים מרובעים [ ].
    * רשימה פשוטה:      ['a', 'b', 'c']
    * רשימה עם שמות:    ['שם' => 'דוד', 'גיל' => 30]
      כאן כל פריט הוא זוג "מפתח => ערך".
    בקוד הזה כל הגדרה של שלוחה היא מערך עם זוגות כאלה, למשל:
        'type' => 'record'    ← המשמעות: type=record בכתובת של השרת

 3) הוספה לסוף מערך
    $steps[] = משהו;   ← מוסיף את "משהו" לסוף הרשימה $steps.

 4) לולאה (foreach)
    foreach ($steps as [$path, $params]) { ... }
    ← "עבור כל פריט ברשימה $steps, תריץ את מה שבתוך הסוגריים המסולסלים".
    בכל סיבוב, $path הוא הנתיב של השלוחה ו-$params הם ההגדרות שלה.

 5) פונקציה (function)
    קטע קוד עם שם, שאפשר להפעיל כמה פעמים בלי לכתוב אותו מחדש.
    function ym(...) { ... }    ← הגדרה (פעם אחת)
    ym('FileAction', [...]);     ← הפעלה (כמה פעמים שרוצים)

 ============================================================================
  איך הקוד עובד בגדול
 ============================================================================
  א. קוראים את הנתונים מהכתובת (טלפון, טוקן, goto).
  ב. בונים רשימה ($steps) של כל השלוחות שצריך ליצור.
     זה החלק שתשנו הכי הרבה - כל שלוחה היא "פריט" ברשימה.
  ג. עוברים על הרשימה בלולאה ושולחים כל שלוחה לשרת.
  ד. יוצרים את הקבצים של ההקראות (TTS) של התפריטים.
  ה. מחזירים למערכת הודעה: הצלחה או כישלון.
*/

// כתובת הבסיס של ה-API. const = ערך קבוע שלא משתנה בהמשך
const YM_BASE = 'https://www.call2all.co.il/ym/api/';


// ===================== פונקציה 1: ym =====================
// התפקיד: לשלוח פקודה כלשהי לשרת ולהחזיר true אם הצליח או false אם נכשל.
//
// מקבלת שני דברים:
//   $action - שם הפקודה (למשל 'UpdateExtension' או 'FileAction')
//   $params - רשימת ההגדרות של הפקודה
//
// : bool בסוף השורה אומר שהפונקציה מחזירה אמת/שקר
function ym(string $action, array $params): bool
{
    // בונים כתובת מלאה. http_build_query הופכת את המערך לטקסט בצורה: key=value&key2=value2
    // היתרון: היא מקודדת לבד תווים מיוחדים (למשל # הופך ל-%23, ועברית נעשית תקינה)
    $url = YM_BASE . $action . '?' . http_build_query($params);

    // שולחים את הבקשה לשרת. ה-@ מסתיר הודעת שגיאה אם החיבור נכשל
    $raw = @file_get_contents($url);

    // אם החיבור נכשל לגמרי - מחזירים "נכשל"
    if ($raw === false) {
        return false;
    }

    // התשובה של השרת היא טקסט בפורמט JSON. הופכים אותה למערך שאפשר לקרוא ממנו
    $res = json_decode($raw, true);

    // מחזירים "הצליח" רק אם השרת ענה responseStatus = OK
    return isset($res['responseStatus']) && $res['responseStatus'] === 'OK';
}


// ===================== פונקציה 2: ext =====================
// התפריט/השלוחות תמיד נשלחים עם אותו token ועם אותו קידומת path.
// במקום לכתוב את זה בכל פעם, הפונקציה הזו מוסיפה אותם לבד.
//
// הסימן + בין שני מערכים אומר "חבר אותם יחד":
//   ['token' => ..., 'path' => ...]  +  $params
// כלומר: ההגדרות שאנחנו שולחים + token + path.
function ext(string $token, string $path, array $params): bool
{
    return ym('UpdateExtension', ['token' => $token, 'path' => 'ivr2:' . $path] + $params);
}

        // הפריט השני: מקש 1 - הוספה לרשימה
